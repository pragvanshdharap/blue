/*
 * Sensor_Logger_UART_Console.c
 * Combined sensor reader, UART CSV logger, and CCES console printer for ADSP-21479
 */

#include <sys/platform.h>
#include "adi_initialize.h"
#include <services/gpio/adi_gpio.h>
#include <drivers/spi/adi_spi.h>
#include <drivers/uart/adi_uart.h>
#include <stdio.h>
#include <math.h>

#define ENCODER_CLK_PORT ADI_GPIO_PORT_C
#define ENCODER_CLK_PIN  ADI_GPIO_PIN_4
#define ENCODER_DT_PORT  ADI_GPIO_PORT_C
#define ENCODER_DT_PIN   ADI_GPIO_PIN_5
#define TILT_PORT        ADI_GPIO_PORT_C
#define TILT_PIN         ADI_GPIO_PIN_7
#define SPI_DEVICE_NUM   0

ADI_SPI_HANDLE hSPI;
ADI_UART_HANDLE hUart;

int encoder_count = 0;
int last_clk_state = 1;

void delay(void) {
    for (volatile int i = 0; i < 100000; ++i);
}

// UART initialization at 115200 baud
void uart_init(void)
{
    static uint8_t uart_memory[ADI_UART_BIDIR_MEMORY_SIZE];
    adi_uart_Open(0, ADI_UART_DIR_BIDIRECTION, uart_memory, sizeof(uart_memory), &hUart);
    adi_uart_SetBaudRate(hUart, ADI_UART_BAUD_115200);
}

// UART CSV output: encoder, tilt, temperature
void uart_send_csv(int encoder, uint8_t tilt, float tempC)
{
    char buf[64];
    int len = sprintf(buf, "%d,%s,%.2f\r\n", encoder, tilt ? "FLAT" : "TILTED", tempC);
    adi_uart_Write(hUart, buf, len, ADI_UART_MODE_BLOCKING);
}

// SPI ADC channel read (e.g., MCP3008 channel 0)
uint16_t read_adc_channel(uint8_t ch) {
    uint8_t tx[3] = {0x01, (0x80 | (ch << 4)), 0x00}, rx[3];
    adi_spi_ReadWrite(hSPI, tx, rx, 3, false);
    return ((rx[1] & 0x03) << 8) | rx[2];
}

// Convert ADC value to temperature (NTC thermistor, 10K)
float convert_ntc_to_temp(uint16_t adc) {
    float v = adc * 3.3f / 1023.0f;
    float r = (v * 10000.0f) / (3.3f - v);
    return 3950.0f / (log(r / 10000.0f) + (3950.0f / (25.0f + 273.15f))) - 273.15f;
}

int main(void) {
    adi_initComponents();
    adi_gpio_Init(NULL, 0);
    uart_init();

    // Configure GPIOs for sensors
    adi_gpio_SetDirection(ENCODER_CLK_PORT, ENCODER_CLK_PIN, ADI_GPIO_DIRECTION_INPUT);
    adi_gpio_SetDirection(ENCODER_DT_PORT, ENCODER_DT_PIN, ADI_GPIO_DIRECTION_INPUT);
    adi_gpio_SetDirection(TILT_PORT, TILT_PIN, ADI_GPIO_DIRECTION_INPUT);

    // SPI Init for external ADC
    adi_spi_Open(SPI_DEVICE_NUM, &hSPI);
    adi_spi_SetBitrate(hSPI, 500000);
    adi_spi_SetClockPolarity(hSPI, false);
    adi_spi_SetClockPhase(hSPI, false);
    adi_spi_SetWordSize(hSPI, 8);
    adi_spi_SetChipSelect(hSPI, ADI_SPI_CS0);

    // Optional: UART CSV header
    char header[] = "Encoder,Tilt,Temperature_C\r\n";
    adi_uart_Write(hUart, header, sizeof(header) - 1, ADI_UART_MODE_BLOCKING);

    while (1) {
        // Rotary encoder logic
        uint8_t clk, dt, tilt;
        adi_gpio_GetData(ENCODER_CLK_PORT, ENCODER_CLK_PIN, &clk);
        adi_gpio_GetData(ENCODER_DT_PORT, ENCODER_DT_PIN, &dt);
        if (clk != last_clk_state) {
            if (dt != clk) encoder_count++;
            else encoder_count--;
        }
        last_clk_state = clk;

        // Read tilt sensor state
        adi_gpio_GetData(TILT_PORT, TILT_PIN, &tilt);

        // Read temperature from ADC
        uint16_t adc = read_adc_channel(0);
        float temp = convert_ntc_to_temp(adc);

        // Print to UART as CSV
        uart_send_csv(encoder_count, tilt, temp);

        // Also print to CCES Console
        printf("[Console] Encoder: %d, Tilt: %s, Temp: %.2f C\n",
               encoder_count,
               tilt ? "FLAT" : "TILTED",
               temp);

        delay();
    }

    return 0;
}

/* Rotary Encoder with M274 on STM32F4 Discovery (STM32F407VGT6)
 * CLK: PA0 (EXTI)
 * DT:  PA1
 * SW:  PA2 (Optional)
 * UART2: PA2 (TX), PA3 (RX)
 */

#include "main.h"
#include <stdio.h>

#define CLK_Pin        GPIO_PIN_0
#define CLK_Port       GPIOA
#define DT_Pin         GPIO_PIN_1
#define DT_Port        GPIOA
#define SW_Pin         GPIO_PIN_2
#define SW_Port        GPIOA

int32_t encoder_count = 0;
extern UART_HandleTypeDef huart2;

int __io_putchar(int ch) {
    HAL_UART_Transmit(&huart2, (uint8_t *)&ch, 1, HAL_MAX_DELAY);
    return ch;
}

void HAL_GPIO_EXTI_Callback(uint16_t GPIO_Pin) {
    if (GPIO_Pin == CLK_Pin) {
        if (HAL_GPIO_ReadPin(DT_Port, DT_Pin) != HAL_GPIO_ReadPin(CLK_Port, CLK_Pin)) {
            encoder_count++; // Clockwise
        } else {
            encoder_count--; // Counter-clockwise
        }
        printf("Encoder Count: %ld\r\n", encoder_count);
    }
}

int main(void)
{
    HAL_Init();
    SystemClock_Config();
    MX_GPIO_Init();
    MX_USART2_UART_Init();

    while (1)
    {
        if (HAL_GPIO_ReadPin(SW_Port, SW_Pin) == GPIO_PIN_RESET) {
            printf("Button Pressed!\r\n");
            HAL_Delay(300);
        }
        HAL_Delay(10);
    }
}

void EXTI0_IRQHandler(void)
{
    HAL_GPIO_EXTI_IRQHandler(GPIO_PIN_0);
}

void MX_GPIO_Init(void)
{
    GPIO_InitTypeDef GPIO_InitStruct = {0};
    __HAL_RCC_GPIOA_CLK_ENABLE();

    GPIO_InitStruct.Pin = CLK_Pin;
    GPIO_InitStruct.Mode = GPIO_MODE_IT_RISING_FALLING;
    GPIO_InitStruct.Pull = GPIO_PULLUP;
    HAL_GPIO_Init(CLK_Port, &GPIO_InitStruct);

    GPIO_InitStruct.Pin = DT_Pin;
    GPIO_InitStruct.Mode = GPIO_MODE_INPUT;
    GPIO_InitStruct.Pull = GPIO_PULLUP;
    HAL_GPIO_Init(DT_Port, &GPIO_InitStruct);

    GPIO_InitStruct.Pin = SW_Pin;
    GPIO_InitStruct.Mode = GPIO_MODE_INPUT;
    GPIO_InitStruct.Pull = GPIO_PULLUP;
    HAL_GPIO_Init(SW_Port, &GPIO_InitStruct);

    HAL_NVIC_SetPriority(EXTI0_IRQn, 0, 0);
    HAL_NVIC_EnableIRQ(EXTI0_IRQn);
}

void MX_USART2_UART_Init(void)
{
    huart2.Instance = USART2;
    huart2.Init.BaudRate = 115200;
    huart2.Init.WordLength = UART_WORDLENGTH_8B;
    huart2.Init.StopBits = UART_STOPBITS_1;
    huart2.Init.Parity = UART_PARITY_NONE;
    huart2.Init.Mode = UART_MODE_TX_RX;
    huart2.Init.HwFlowCtl = UART_HWCONTROL_NONE;
    huart2.Init.OverSampling = UART_OVERSAMPLING_16;
    HAL_UART_Init(&huart2);
}

int main(void)
{
  HAL_Init();
  SystemClock_Config();
  MX_GPIO_Init();
  MX_USART2_UART_Init();

  char uart_buf[64];
  uint32_t timestamp = 0;

  send_uart("Time(ms),Tilt_Status\r\n");  // CSV header

  while (1)
  {
    GPIO_PinState tilt = HAL_GPIO_ReadPin(GPIOA, GPIO_PIN_0);

    if (tilt == GPIO_PIN_RESET) {
      HAL_GPIO_WritePin(GPIOD, GPIO_PIN_12, GPIO_PIN_SET);  // LED ON
      snprintf(uart_buf, sizeof(uart_buf), "%lu,TILT\r\n", timestamp);
    } else {
      HAL_GPIO_WritePin(GPIOD, GPIO_PIN_12, GPIO_PIN_RESET); // LED OFF
      snprintf(uart_buf, sizeof(uart_buf), "%lu,NO_TILT\r\n", timestamp);
    }

    send_uart(uart_buf);
    HAL_Delay(500);
    timestamp += 500;
  }
}
